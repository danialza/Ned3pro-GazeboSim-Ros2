Wrapper Code which works : 

python3 - <<'PY'
import math, time
import rospy
from niryo_robot_python_ros_wrapper import NiryoRosWrapper

rospy.init_node('niryo_wrapper_cli', anonymous=True)
robot = NiryoRosWrapper()

t0 = time.time()
rate = rospy.Rate(2)  # 2Hz برای تست (نه 20Hz)

while not rospy.is_shutdown():
    t = time.time() - t0
    j = [
        0.5 * math.sin(t),
        0.5 * math.sin(t + 1.0),
        -1.0 + 0.3 * math.sin(t + 2.0),
        0.3 * math.sin(t + 3.0),
        0.3 * math.sin(t + 4.0),
        0.3 * math.sin(t + 5.0),
    ]
    robot.move_joints(*j)
    rate.sleep()
PY
